\[\begin{array}{@{}l@{}}
  \text{\color{acc}measure} \\[5pt]
  \quad \ell = \len(u,\, a)
    \quad {\color{dim}\text{the angle length from } u \text{ to what arrived}} \\[5pt]
  \quad R = \text{the lengths this mode rests at}
    \quad {\color{dim}\text{a list of numbers}} \\[5pt]
  \quad f(u) = \text{the smallest } \lvert \ell - r \rvert \text{ over the } r \text{ in } R
    \\[9pt]
  \text{\color{acc}then decide} \\[5pt]
  \quad \text{if } f(t) = 0 : \qquad\qquad\;\; \text{stay at } t \\[5pt]
  \quad \text{else if } f(t_{+1}) \le f(t_{-1}) : \quad \text{go to } t_{+1} \\[5pt]
  \quad \text{else} : \qquad\qquad\qquad\quad\;\; \text{go to } t_{-1}
\end{array}\]
