\[\begin{array}{@{}l@{}}
  \text{\color{acc}directions} \quad \mathbb{Z}_{24},\;\text{generated by } +1
    \qquad t \mapsto t \pm 1 \;\text{is the turn} \\[5pt]
  \text{\color{acc}lines} \quad \mathbb{Z}_{12} = \mathbb{Z}_{24} / \langle 12 \rangle
    \qquad t \text{ and } t+12 \text{ are the same line} \\[10pt]
  \text{distance} \quad L(x,y) = \min\bigl(\lvert x-y \rvert,\; 24 - \lvert x-y \rvert\bigr)
    \quad \text{— the word metric on } \mathbb{Z}_{24} \\[5pt]
  \text{what arrives} \quad a = p + 6 \quad \text{— the partner's normal} \\[10pt]
  \text{let } g = t - p \quad \text{— the two tilts, as directions} \\[5pt]
  \qquad {\color{par}\text{parallel} \iff g \equiv 0 \pmod{12}}
  \qquad {\color{perp}\text{perpendicular} \iff g \equiv 6 \pmod{12}} \\[10pt]
  \text{so } L(t,a) = \text{fold}\lvert g - 6 \rvert
    \quad\Longrightarrow\quad
    {\color{par}g \equiv 0 \Rightarrow L = 6}
    \quad
    {\color{perp}g \equiv 6 \Rightarrow L \in \{0,\,12\}}
\end{array}\]
